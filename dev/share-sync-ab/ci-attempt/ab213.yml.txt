name: ab-213

on:
  push:
    branches: ["ab/213-*"]

permissions:
  contents: read

# A new push to a pull request replaces its running checks; main keeps every run.
concurrency:
  group: ab213-${{ github.event.pull_request.number || github.sha }}
  cancel-in-progress: ${{ github.event_name == 'pull_request' }}

# Set before every Rust cache lookup; dependency/codec opt-level overrides stay in Cargo.toml.
env:
  CARGO_PROFILE_DEV_DEBUG: "0"
  CARGO_PROFILE_TEST_DEBUG: "0"
  CARGO_INCREMENTAL: "0"

jobs:
  native-build:
    runs-on: macos-15
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1

        with:
          toolchain: stable
      - run: brew install opus
      - name: Fixture and native bridge
        run: |
          cargo build --locked -p gwar-client --no-default-features --example native_bridge --example web_latency
          mkdir -p "$RUNNER_TEMP/native/target/debug/examples"
          cp target/debug/examples/native_bridge target/debug/examples/web_latency "$RUNNER_TEMP/native/target/debug/examples/"
          tar -C "$RUNNER_TEMP/native" -czf "$RUNNER_TEMP/native-browser.tar.gz" target
      - uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: native-browser-binaries
          path: ${{ runner.temp }}/native-browser.tar.gz
          if-no-files-found: error
          retention-days: 1

  # Exact finished-output caches; validation still runs on every hit.
  mls-wasm:
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        timeout-minutes: 5
      - uses: dtolnay/rust-toolchain@683b588c487a69d01f6c6ddf40f036ff682cc2c8 # v1
        timeout-minutes: 5
        with:
          toolchain: 1.99.0
          targets: wasm32-unknown-unknown
      - uses: pnpm/action-setup@ea17c68df8912ef543352723c149a84f56e3d413 # v6.1.0
        timeout-minutes: 5
      - uses: actions/setup-node@949feb2413d6458794dcd2491c4babbbce0c15c1 # v7.1.0
        timeout-minutes: 5
        with:
          node-version: 22.23.3
          cache: pnpm
      - name: Exact wasm output key
        timeout-minutes: 5
        id: wasm-key
        run: python3 scripts/ci/wasm-key.py mls
      - name: Restore finished wasm
        timeout-minutes: 5
        id: pkg
        uses: actions/cache/restore@55cc8345863c7cc4c66a329aec7e433d2d1c52a9 # v6.1.0
        with:
          path: crates/gwar-mls-wasm/pkg/
          key: ${{ steps.wasm-key.outputs.key }}
      - run: pnpm install --frozen-lockfile
        timeout-minutes: 5
      - name: Pinned wasm tools
        timeout-minutes: 12
        if: steps.pkg.outputs.cache-hit != 'true'
        run: |
          cargo install wasm-bindgen-cli --version 0.2.129 --locked
          npm install --prefix .build-tools --no-audit --no-fund binaryen@123.0.0
      - name: Build artifact path regression
        timeout-minutes: 5
        run: node scripts/mls-wasm/test-build.mjs
      - name: Build production wasm on cache miss
        timeout-minutes: 15
        if: steps.pkg.outputs.cache-hit != 'true'
        env:
          WASM_OPT: .build-tools/node_modules/.bin/wasm-opt
        run: scripts/mls-wasm/build.sh
      - name: Production wasm gzip budget (also on hits)
        timeout-minutes: 5
        run: node scripts/mls-wasm/size.mjs
      - name: JavaScript agent smoke test
        timeout-minutes: 5
        run: node scripts/mls-wasm/smoke.mjs
      - uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        timeout-minutes: 5
        with:
          name: mls-wasm
          path: crates/gwar-mls-wasm/pkg/
          if-no-files-found: error
          retention-days: 7

      - name: Save validated wasm output
        timeout-minutes: 5
        if: steps.pkg.outputs.cache-hit != 'true'
        uses: actions/cache/save@55cc8345863c7cc4c66a329aec7e433d2d1c52a9 # v6.1.0
        with:
          path: crates/gwar-mls-wasm/pkg/
          key: ${{ steps.wasm-key.outputs.key }}
